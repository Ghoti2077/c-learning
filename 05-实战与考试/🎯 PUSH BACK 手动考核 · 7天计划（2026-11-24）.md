```cpp
#include "vex.h"
#include <cmath>
using namespace vex;

vex::brain Brain;
controller Controller = controller(primary);
// ……left_motor_1 … right_motor_5、left_drive、right_drive、imu、hook、mark、
// ……intake_motor、intake1~3、outputmotor、sortmotor、sort 全部照原样放着

// ===== 线程之间共享的数据（谁写谁读，写在这里最显眼）=====
double dist_in   = 0;   // 底盘累计里程（英寸）
double head_deg  = 0;   // 航向（0~360）
double temp_l    = 0;   // 左侧最高温（℃）
double temp_r    = 0;   // 右侧最高温（℃）

// ===== 小工具 =====
double hottest(double a, double b, double c, double d, double e) {
  double m = a;
  if (b > m) m = b;
  if (c > m) m = c;
  if (d > m) m = d;
  if (e > m) m = e;
  return m;
}

// 轮径 2.75 in、外齿 43/50 → 电机转 1 圈轮子走 7.4299 in
const double IN_PER_DEG = 3.14159265 * 2.75 * (43.0 / 50.0) / 360.0;  // 0.020639

void update_distance() {
  static double last_l = 0, last_r = 0;
  double l = left_motor_1.rotation(degrees);
  double r = right_motor_1.rotation(degrees);
  dist_in += ((l - last_l) + (r - last_r)) / 2.0 * IN_PER_DEG;  // 两侧取平均 = 车体中心
  last_l = l;
  last_r = r;
}
```

```cpp
void chassis_control() {
  float straight = Controller.Axis3.position();
  float turn     = Controller.Axis1.position();
  if (fabsf(straight) <= 5.0f) straight = 0.0f;   // 死区
  if (fabsf(turn)     <= 5.0f) turn     = 0.0f;
  if (straight == 0.0f && turn == 0.0f) {
    left_drive.stop(coast);
    right_drive.stop(coast);
    return;
  }
  float l = straight + turn * 0.4f;
  float r = straight - turn * 0.4f;
  left_drive.spin(fwd,  l / 100.0f * 12.0f, volt);
  right_drive.spin(fwd,  r / 100.0f * 12.0f, volt);
}
```

```cpp
void display() {                                  // 这个函数整个跑在独立线程里
  while (true) {
    temp_l = hottest(left_motor_1.temperature(temperatureUnits::celsius), /* …5 台 */);
    temp_r = hottest(right_motor_1.temperature(temperatureUnits::celsius), /* …5 台 */);
    head_deg = imu.heading(degrees);

    Controller.Screen.setCursor(1, 1);
    Controller.Screen.print("L%3.0f R%3.0f", temp_l, temp_r);   // 第 1 行：温度
    Controller.Screen.setCursor(2, 1);
    Controller.Screen.print("D %6.1f in", dist_in);             // 第 2 行：里程
    Controller.Screen.setCursor(3, 1);
    Controller.Screen.print("H %5.0f deg", head_deg);           // 第 3 行：航向
    wait(100, msec);
  }
}
```

```cpp
void hook_control() {
  static bool last = false;      // 上一轮按没按
  static bool on   = false;      // 钩子现在抬没抬
  bool now = Controller.ButtonR2.pressing();
  if (now && !last) {            // 只在「刚按下去」这一瞬间翻一次
    on = !on;
    hook.set(on);
  }
  last = now;
}

int main() {

    update_distance();
```

| 11 | 吐球函数里顺手 `stop()` 吸球电机会和吸球按键打架 | —— | 每个电机只由**一个**函数负责开关 |
